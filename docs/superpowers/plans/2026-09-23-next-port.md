# Prologue& Next.js 이식 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 시안(`design/mockups/v3`)의 Projects 목록·Case Study 상세·문의 서랍을 `web/`의 Next.js 앱으로 옮기고 Vercel에 배포한다.

**Architecture:** Next 15 App Router. 페이지는 서버 컴포넌트로 정적 생성하고, 움직이는 부분(판 엔진·핫스팟·morph·와이프·콜라주·문의 서랍·전환 링크)만 클라이언트 컴포넌트로 분리한다. 데이터는 `content/`의 TypeScript 파일, 화면은 `public/screens/`의 PNG. 문의는 Route Handler가 받아 Gmail SMTP로 발송한다.

**Tech Stack:** Next 15 (App Router) · TypeScript · CSS Modules · zod · nodemailer · Playwright · next/font

**Spec:** `docs/superpowers/specs/2026-09-23-next-port-design.md`

## Global Constraints

- 작업 디렉터리는 `web/`. 저장소의 `design/`·`docs/`·`data/`는 읽기만 하고 수정하지 않는다.
- 판·격자 수치는 시안 값을 그대로 쓴다: 컨테이너 `max-width: clamp(1272px, 80vw, 1760px)`, 좌우 여백 `72px`, 헤더 높이 `64px`, 판 높이 `calc(100vh - 64px)`(하한 `640px`), 판 상단 여백 `clamp(28px, 5vh, 56px)`(표지 `clamp(80px, 9vh, 104px)`), 판 하단 여백 `40px`.
- 엔진 값: `smooth 0.28`, 휠 `{gap:100, first:90, more:480}`, 이동 잠금 `160ms`, 머무름 구간 `120vh`, 붙는 위치 `top 64px`.
- 그림 비율: 다이어그램 `640:360`, 화면 `1280:800`, 폰 `390:844`, 표지 화면은 판 높이의 `0.62` 이하.
- 색 토큰은 `design/mockups/v3/v3.css`의 `:root` 값을 그대로 복사한다(navy-950 `#151A33` … bone-900 `#1C1B18`, `--signal:#3B6FE0`, `--ease-out:cubic-bezier(.2,.8,.2,1)`).
- 글꼴: 제목·로고 `Bodoni Moda`(serif), 본문 `Pretendard`(sans).
- 문구는 docs/21 규칙(번역체·줄표·"당신" 금지). 새 문구를 지어내지 말고 시안 `cases.js`의 문자열을 그대로 옮긴다.
- 엔진은 클래스가 아니라 `data-*` 속성으로 요소를 찾는다.
- 커밋 메시지 접두사 `[ADD]/[MOD]/[FIX]/[DOCS]`, 본문 한국어, 끝에 `Co-Authored-By: Claude Opus 5 (1M context) <noreply@anthropic.com>`.

```tsx
  metadataBase: new URL(site),

  description: '복잡한 업무를 단순한 제품으로 바꿉니다.',

  return (

    </html>
  );
```

```ts
import { projects, PROBLEM_NAME } from './projects';
import { porFavorHarry } from './cases/por-favor-harry';
import type { Case, Project } from './types';

export const cases: Record<string, Case> = { 'por-favor-harry': porFavorHarry };
export const allProjects = (): Project[] => [...projects].sort((a, b) => a.order - b.order);
export const publishedProjects = (): Project[] => allProjects().filter((p) => p.published);
export const getProject = (slug: string): Project | undefined => projects.find((p) => p.slug === slug);
export const getCase = (slug: string): Case | undefined => cases[slug];
export const nextProject = (slug: string): Project | undefined =>
  publishedProjects().filter((p) => p.slug !== slug)[0];
export { PROBLEM_NAME };
export type { Case, Project, Pic, Spot, Scatter } from './types';
```

```tsx
'use client';
import Link from 'next/link';
import { useRouter } from 'next/navigation';
import type { MouseEvent, PointerEvent, ReactNode } from 'react';

type Props = { href: string; className?: string; children: ReactNode; shareTitle?: boolean; 'data-row'?: boolean };

const NAME = 'pj-title';

export function ViewTransitionLink({ href, className, children, shareTitle, ...rest }: Props) {
  const router = useRouter();

  // 클릭 직전에 이 행의 제목에만 이름을 붙인다 — 문서에 하나만 있어야 전환이 성립한다
  const mark = (e: PointerEvent<HTMLAnchorElement>) => {
    if (!shareTitle) return;
    document.querySelectorAll<HTMLElement>('[data-title]').forEach((el) => { el.style.viewTransitionName = ''; });
    const t = e.currentTarget.querySelector<HTMLElement>('[data-title]');
    if (t) t.style.viewTransitionName = NAME;
  };

  const go = (e: MouseEvent<HTMLAnchorElement>) => {
    if (e.metaKey || e.ctrlKey || e.shiftKey || e.button !== 0) return;
    const doc = document as Document & { startViewTransition?: (cb: () => void) => { finished: Promise<void> } };
    if (!doc.startViewTransition) return;              // 미지원 브라우저는 기본 이동
    e.preventDefault();
    const vt = doc.startViewTransition(() => { router.push(href); });
    vt.finished.finally(() => {
      document.querySelectorAll<HTMLElement>('[data-title]').forEach((el) => { el.style.viewTransitionName = ''; });
    });
  };

  return <Link href={href} className={className} onPointerDown={mark} onClick={go} {...rest}>{children}</Link>;
}
```

```ts
export function useFitPans() {

    const once = () => {
      document.querySelectorAll<HTMLElement>('[data-pan]').forEach((pan) => {

        if (!target) return;
        if (!desktop()) { target.style.removeProperty('--dw'); pan.style.removeProperty('--tw'); return; }

        const ratio = Number(target.dataset.ratio ?? 1280 / 800);  // 폭/높이
        const cap = box.querySelector<HTMLElement>('[data-cap]');

        let h: number;
        if (pan.dataset.pan === 'cover') {
          const title = pan.querySelector<HTMLElement>('[data-cover-title]');
          h = Math.min(pan.clientHeight * 0.62, pan.clientHeight - padT - 60 - (title?.offsetHeight ?? 0)) - capH;
        } else {
          const txt = pan.querySelector<HTMLElement>('[data-text]');
          h = pan.clientHeight - padT - 40 - (txt?.offsetHeight ?? 0) - capH;
        }
        const w = clamp(h * ratio, pan.dataset.pan === 'phone' ? 150 : 280, box.clientWidth);
        target.style.setProperty('--dw', `${Math.round(w)}px`);
        pan.style.setProperty('--tw', pan.dataset.pan === 'phone' ? `${Math.min(box.clientWidth, 704)}px` : `${Math.round(w)}px`);
      });
    };

    const run = () => { once(); once(); };   // 문단 폭 ↔ 그림 폭이 서로 영향을 주므로 두 번
    run();
    document.fonts?.ready.then(run);
    const ro = new ResizeObserver(run);
    ro.observe(document.body);
    window.addEventListener('resize', run);
    return () => { ro.disconnect(); window.removeEventListener('resize', run); };
```

```ts
'use client';
import { useEffect, useState, type RefObject } from 'react';

const clamp = (v: number, a: number, b: number) => Math.max(a, Math.min(b, v));

/** 구간(120vh) 안에서 판을 top 64 에 붙이고 진행률 t 를 돌려준다 */
export function useDwell(ref: RefObject<HTMLElement | null>, steps: 2 | 3): number {
  const [t, setT] = useState(0);
  useEffect(() => {
    const el = ref.current;
    if (!el) return;
    const reduced = window.matchMedia('(prefers-reduced-motion:reduce)').matches;
    const desktop = () => window.matchMedia('(min-width:1024px)').matches;

    const onScroll = () => {
      const pan = el.querySelector<HTMLElement>('[data-pan]');
      if (!pan) return;
      if (!desktop()) { pan.style.transform = ''; setT(1); return; }
      const r = el.getBoundingClientRect();
      const range = Math.max(1, r.height - pan.offsetHeight);
      const off = clamp(64 - r.top, 0, range);
      pan.style.transform = off ? `translate3d(0, ${off.toFixed(1)}px, 0)` : '';
      const raw = off / range;
      setT(reduced ? (raw >= 0.5 ? 1 : 0) : raw);
    };

    onScroll();
    window.addEventListener('scroll', onScroll, { passive: true });
    window.addEventListener('resize', onScroll);
    return () => { window.removeEventListener('scroll', onScroll); window.removeEventListener('resize', onScroll); };
  }, [ref, steps]);
  return t;
}
```

- [ ] **Step 3: useStations**

```ts
'use client';
import { useCallback, useEffect, useRef } from 'react';

const clamp = (v: number, a: number, b: number) => Math.max(a, Math.min(b, v));
const WHEEL = { gap: 100, first: 90, more: 480 };
const SMOOTH = 0.28;
const LOCK = 160;

function smoothDamp(cur: number, to: number, vel: number, smoothTime: number, dt: number) {
  const w = 2 / Math.max(0.0001, smoothTime), x = w * dt;
  const e = 1 / (1 + x + 0.48 * x * x + 0.235 * x * x * x);
  const ch = cur - to, tt = (vel + w * ch) * dt;
  return { pos: to + (ch + tt) * e, vel: (vel - w * tt) * e };
}

export function useStations() {
  const anim = useRef({ pos: 0, target: 0, vel: 0, active: false });

  const stops = useCallback((): number[] => {
    const out: number[] = [];
    document.querySelectorAll<HTMLElement>('[data-pan]').forEach((pan) => {
      const dwell = pan.closest<HTMLElement>('[data-dwell]');
      if (!dwell) { out.push(pan.getBoundingClientRect().top + window.scrollY - 64); return; }
      const y0 = dwell.getBoundingClientRect().top + window.scrollY;
      const range = dwell.offsetHeight - pan.offsetHeight;
      const ks = dwell.dataset.dwell === '2' ? [0, 1] : [0, 0.5, 1];
      ks.forEach((k) => out.push(y0 + range * k));
    });
    return out.sort((a, b) => a - b);
  }, []);

  const setTarget = useCallback((y: number) => {
    const max = document.documentElement.scrollHeight - window.innerHeight;
    anim.current.target = clamp(y, 0, max);
    if (!anim.current.active) { anim.current.pos = window.scrollY; anim.current.vel = 0; anim.current.active = true; }
  }, []);

  const goStop = useCallback((dir: 1 | -1) => {
    const st = stops();
    let k = 0;
    st.forEach((y, i) => { if (y <= (anim.current.active ? anim.current.target : window.scrollY) + 2) k = i; });
    setTarget(st[clamp(k + dir, 0, st.length - 1)]);
  }, [stops, setTarget]);

  const goTo = useCallback((el: HTMLElement, immediate = false) => {
    const dwell = el.closest<HTMLElement>('[data-dwell]');
    const y = (dwell ?? el).getBoundingClientRect().top + window.scrollY - (dwell ? 0 : 64);
    if (immediate) { anim.current.active = false; window.scrollTo(0, y); } else setTarget(y);
  }, [setTarget]);

  useEffect(() => {
    if (!window.matchMedia('(min-width:1024px)').matches) return;
    if (window.matchMedia('(prefers-reduced-motion:reduce)').matches) return;
    history.scrollRestoration = 'manual';

    let last = performance.now(), raf = 0;
    const tick = (now: number) => {
      const dt = clamp((now - last) / 1000, 0.001, 0.05); last = now;
      const a = anim.current;
      if (a.active) {
        const r = smoothDamp(a.pos, a.target, a.vel, SMOOTH, dt);
        a.pos = r.pos; a.vel = r.vel;
        if (Math.abs(a.target - a.pos) < 0.5 && Math.abs(a.vel) < 8) { a.pos = a.target; a.vel = 0; a.active = false; }
        window.scrollTo(0, a.pos);
      }
      raf = requestAnimationFrame(tick);
    };
    raf = requestAnimationFrame(tick);

    let acc = 0, lastEv = 0, stepped = false, streamStart = 0, lockUntil = 0;
    const onWheel = (e: WheelEvent) => {
      if (e.ctrlKey) return;                                   // 브라우저 확대는 건드리지 않는다
      if (document.querySelector('[data-drawer-open]')) return;
      e.preventDefault();
      const d = e.deltaY;
      if (!d) return;
      const now = performance.now(), gap = now - lastEv; lastEv = now;
      if (gap > WHEEL.gap) { acc = 0; stepped = false; streamStart = now; }
      acc += d;
      const fire = (dir: 1 | -1) => { if (now < lockUntil) return; goStop(dir); lockUntil = now + LOCK; };
      if (!stepped) { if (Math.abs(acc) >= WHEEL.first) { fire(acc > 0 ? 1 : -1); acc = 0; stepped = true; } return; }
      if (now - streamStart > 300 && Math.abs(acc) >= WHEEL.more) { fire(acc > 0 ? 1 : -1); acc = 0; }
    };

    const onKey = (e: KeyboardEvent) => {
      if (e.altKey || e.ctrlKey || e.metaKey) return;
      if (document.querySelector('[data-drawer-open]')) return;
      if (['ArrowDown', 'ArrowRight', 'PageDown', ' '].includes(e.key)) { e.preventDefault(); goStop(1); }
      else if (['ArrowUp', 'ArrowLeft', 'PageUp'].includes(e.key)) { e.preventDefault(); goStop(-1); }
      else if (e.key === 'Home') { e.preventDefault(); setTarget(0); }
      else if (e.key === 'End') { e.preventDefault(); setTarget(document.documentElement.scrollHeight - window.innerHeight); }
    };

    window.addEventListener('wheel', onWheel, { passive: false });
    window.addEventListener('keydown', onKey);
    return () => {
      cancelAnimationFrame(raf);
      window.removeEventListener('wheel', onWheel);
      window.removeEventListener('keydown', onKey);
    };
  }, [goStop, setTarget]);

  return { goStop, goTo, stops };
}
```

```ts
import nodemailer from 'nodemailer';
import type { InquiryInput } from './inquiry-schema';

const LABEL: Record<string, string> = {
  work: '지금 하는 업무', tools: '쓰는 도구', pain: '가장 불편한 점', people: '사용 인원',
  repeat: '반복 주기', kind: '개선/신규', goal: '원하는 결과', when: '희망 일정',
  budget: '예산 범위', email: '연락처', project: '보고 온 프로젝트',
};
const KIND: Record<string, string> = { improve: '기존 시스템 개선', new: '신규 구축', unsure: '아직 모름' };

export async function sendInquiry(input: InquiryInput): Promise<void> {
  const user = process.env.GMAIL_USER, pass = process.env.GMAIL_APP_PASSWORD;
  const to = process.env.INQUIRY_TO ?? user;
  if (!user || !pass || !to) throw new Error('메일 환경변수가 없습니다');

  const lines = (['work', 'tools', 'pain', 'people', 'repeat', 'kind', 'goal', 'when', 'budget', 'email', 'project'] as const)
    .map((k) => {
      const v = k === 'kind' ? KIND[input.kind] : String(input[k] ?? '');
      return v ? `${LABEL[k]}\n${v}\n` : '';
    })
    .filter(Boolean)
    .join('\n');

  const transport = nodemailer.createTransport({ service: 'gmail', auth: { user, pass } });
  await transport.sendMail({
    from: `Prologue& <${user}>`,
    to,
    replyTo: input.email,
    subject: `[문의] ${input.pain.slice(0, 30)}${input.project ? ` · ${input.project}` : ''}`,
    text: lines,
  });
}
```

```ts
import { NextResponse } from 'next/server';
import { inquirySchema } from '@/lib/inquiry-schema';
import { sendInquiry } from '@/lib/mail';

export async function POST(req: Request) {
  let body: unknown;
  try { body = await req.json(); } catch { return NextResponse.json({ ok: false, error: '본문을 읽지 못했습니다' }, { status: 400 }); }

  const parsed = inquirySchema.safeParse(body);
  if (!parsed.success) {
    const first = parsed.error.issues[0]?.message ?? '입력을 확인해주세요';
    return NextResponse.json({ ok: false, error: first }, { status: 400 });
  }
  const data = parsed.data;

  // 봇: 허니팟이 채워졌거나 5초 안에 제출 → 조용히 성공 처리하고 보내지 않는다
  if (data.website || data.elapsed < 5000) return NextResponse.json({ ok: true, sent: false });

  try {
    await sendInquiry(data);
    return NextResponse.json({ ok: true, sent: true });
  } catch (e) {
    console.error('inquiry send failed', e);
    return NextResponse.json({ ok: false, error: '보내지 못했습니다' }, { status: 502 });
  }
}
```
